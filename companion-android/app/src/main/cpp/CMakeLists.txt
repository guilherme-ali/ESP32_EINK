cmake_minimum_required(VERSION 3.22.1)
project(voice_opus C)
if(NOT DEFINED OPUS_SOURCE_DIR)
    message(FATAL_ERROR "Use build_android.py to authenticate the official libopus source")
endif()
include(CheckSymbolExists)
set(CMAKE_REQUIRED_LIBRARIES m)
check_symbol_exists(lrint math.h HAVE_LRINT)
check_symbol_exists(lrintf math.h HAVE_LRINTF)
set(OPUS_FILES)
foreach(MANIFEST opus_sources.mk celt_sources.mk silk_sources.mk)
    file(READ "${OPUS_SOURCE_DIR}/${MANIFEST}" CONTENT)
    string(REPLACE "\\\n" " " CONTENT "${CONTENT}")
    foreach(GROUP OPUS_SOURCES CELT_SOURCES SILK_SOURCES SILK_SOURCES_FIXED)
        string(REGEX MATCH "(^|\n)${GROUP} =[^\n]*" LINE "${CONTENT}")
        if(LINE)
            string(REGEX REPLACE "^[\n]*${GROUP} = *" "" LINE "${LINE}")
            separate_arguments(FILES UNIX_COMMAND "${LINE}")
            foreach(FILE ${FILES})
                list(APPEND OPUS_FILES "${OPUS_SOURCE_DIR}/${FILE}")
            endforeach()
        endif()
    endforeach()
endforeach()
add_library(voice_opus SHARED voice_opus.c ${OPUS_FILES})
target_include_directories(voice_opus PRIVATE "${OPUS_SOURCE_DIR}/include" "${OPUS_SOURCE_DIR}" "${OPUS_SOURCE_DIR}/celt" "${OPUS_SOURCE_DIR}/silk" "${OPUS_SOURCE_DIR}/silk/fixed")
target_compile_definitions(voice_opus PRIVATE OPUS_BUILD FIXED_POINT DISABLE_FLOAT_API VAR_ARRAYS)
if(HAVE_LRINT)
    target_compile_definitions(voice_opus PRIVATE HAVE_LRINT)
endif()
if(HAVE_LRINTF)
    target_compile_definitions(voice_opus PRIVATE HAVE_LRINTF)
endif()
target_compile_options(voice_opus PRIVATE -O2 -fno-strict-aliasing)
target_link_libraries(voice_opus m)
